    {Design of a controller for prediction and correction of gait cycle} % Event
    {University at Buffalo} % Location
    {Jan. 2017 - Dec. 2017} % Date(s)
    {
      \begin{cvitems} % Description(s)
        \item {Design and construct a non-linear actuator system for gait correction that utilizes both stimulation of muscles and actuators.}
        \item {Determine and quantify key performance parameters and compare them to commercially available systems.}
        \item{Developed an intuitive gait \emph{labeling algorithm} using joint-angle data which permits very high classification accuracy.}
        \item{Achieved a 99.4\% binary \emph{classification accuracy} and 98.68\% \emph{Cohen's Kappa} using K-Means clustering in combined EMG gait data from three subjects. Testing was based on 10-fold \emph{cross validation}.}
      \end{cvitems}
    }
